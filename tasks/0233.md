let pending_flags = self.signals.dirty_flags;

if (pending_flags & SIGNAL_TOPOLOGY_CHANGED) != 0 {
    let new_max = self.topology.max_geometry_cells(1, 1) as u32;
    self.signals.active_agent_count = self.signals.active_agent_count.min(new_max);
}

if (pending_flags & SIGNAL_CONSENSUS_SHIFT) != 0 {
    let global_phi = crate::PHI_ANCHOR_CHAIN.lock().global_phi();
    for i in 0..self.signals.active_agent_count as usize {
        let agent = &mut *self.minimal_agents_ptr.add(i);
        agent.phase = agent.phase.wrapping_add(global_phi) & max_phase;
    }
}

self.signals.dirty_flags &= !(SIGNAL_TOPOLOGY_CHANGED | SIGNAL_CONSENSUS_SHIFT);

- **Філософія виправлення:** PoUW не може доводити закон, який ядро вже
  відкинуло.

let net_burn = metabolic_burn.saturating_sub(resonance_credit).max(1);
let mut new_energy = agent.energy as i32 - net_burn as i32;
