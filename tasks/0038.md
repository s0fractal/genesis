# 🧠 1. Формальне ядро (те, чого у тебе нема)

Стан:

```ts
type State = {
  theta: Float32Array; // фаза [0, 2π)
  omega: Float32Array; // власна частота
  energy: Float32Array; // енергія вузла
};
```

І один закон еволюції:

---

## 🔹 Kuramoto (реальний, не декоративний)

\frac{d\theta_i}{dt} = \omega_i + \frac{K}{N} \sum_{j=1}^{N} \sin(\theta_j -
\theta_i)

---

І другий:

---

## 🔹 Закон енергії

E_i(t+1) = E_i(t) + \alpha \cdot R_i - \beta \cdot E_i(t)

де

- (R_i) — локальна синхронізація (resonance)
- (\alpha) — підсилення
- (\beta) — дисипація

---

# ⚙️ 2. Реальний код ядра (мінімальний)

Ось це — вже система. Не симулякр.

```ts
const N = 1024;

const theta = new Float32Array(N);
const omega = new Float32Array(N);
const energy = new Float32Array(N);

const K = 2.0;
const alpha = 0.1;
const beta = 0.01;
const dt = 0.016;

// init
for (let i = 0; i < N; i++) {
  theta[i] = Math.random() * Math.PI * 2;
  omega[i] = (Math.random() - 0.5) * 0.5;
  energy[i] = 1.0;
}
```

---

## 🔁 Один крок еволюції

```ts
function step() {
  const newTheta = new Float32Array(N);
  const newEnergy = new Float32Array(N);

  for (let i = 0; i < N; i++) {
    let coupling = 0;

    for (let j = 0; j < N; j++) {
      coupling += Math.sin(theta[j] - theta[i]);
    }

    coupling = (K / N) * coupling;

    const dTheta = omega[i] + coupling;
    newTheta[i] = theta[i] + dTheta * dt;

    // локальна синхронізація як енергія
    const resonance = Math.abs(coupling);

    newEnergy[i] = energy[i] +
      alpha * resonance -
      beta * energy[i];
  }

  theta.set(newTheta);
  energy.set(newEnergy);
}
```

### ✔ Інваріант 3 — порядок (order parameter)

R = \frac{1}{N} \left| \sum_{j=1}^{N} e^{i\theta_j} \right|

```ts
function orderParameter(): number {
  let re = 0;
  let im = 0;

  for (let i = 0; i < N; i++) {
    re += Math.cos(theta[i]);
    im += Math.sin(theta[i]);
  }

  return Math.sqrt(re * re + im * im) / N;
}
```

👉 це **реальний сигнал**, а не "entropy = 1.5"

# ⚙️ 2. Крок еволюції (CPU reference, truth layer)

Це **еталон**, з яким ти будеш звіряти GPU.

```rust
pub fn step(field: &mut PhaseField, dt: f32, k: f32, alpha: f32, beta: f32) {
    let n = field.n;

    let mut new_theta = vec![0.0; n];
    let mut new_energy = vec![0.0; n];

    for i in 0..n {
        let mut coupling = 0.0;

        for j in 0..n {
            coupling += (field.theta[j] - field.theta[i]).sin();
        }

        coupling *= k / n as f32;

        let dtheta = field.omega[i] + coupling;
        new_theta[i] = field.theta[i] + dtheta * dt;

        let resonance = coupling.abs();

        new_energy[i] = field.energy[i]
            + alpha * resonance
            - beta * field.energy[i];

        if new_energy[i] < 0.0 {
            new_energy[i] = 0.0;
        }
    }

    field.theta = new_theta;
    field.energy = new_energy;
}
```

@group(0) @binding(0) var<storage, read> theta: array<f32>;
@group(0) @binding(1) var<storage, read> omega: array<f32>;
@group(0) @binding(2) var<storage, read> energy: array<f32>;

// x = dt, y = K, z = alpha, w = beta

@compute @workgroup_size(64)
fn main(@builtin(global_invocation_id) id: vec3<u32>) {
    let i = id.x;

    var coupling: f32 = 0.0;
    let n: u32 = arrayLength(&theta);

    for (var j: u32 = 0u; j < n; j = j + 1u) {
        coupling += sin(theta[j] - theta[i]);
    }

    coupling = (params.y / f32(n)) * coupling;

    let dtheta = omega[i] + coupling;
    out_theta[i] = theta[i] + dtheta * params.x;

    let resonance = abs(coupling);

    var e = energy[i]
        + params.z * resonance
        - params.w * energy[i];

    if (e < 0.0) {
        e = 0.0;
    }

    out_energy[i] = e;
}
```

    let t_off = meta.theta_offset;
    let o_off = meta.omega_offset;
    let e_off = meta.energy_offset;

    let theta_i = data[t_off + i];
    let omega_i = data[o_off + i];
    let energy_i = data[e_off + i];

    var coupling: f32 = 0.0;

    for (var j: u32 = 0u; j < meta.n; j = j + 1u) {
        let theta_j = data[t_off + j];
        coupling += sin(theta_j - theta_i);
    }

    coupling = (params.y / f32(meta.n)) * coupling;

    let dtheta = omega_i + coupling;
    let new_theta = theta_i + dtheta * params.x;

    let resonance = abs(coupling);

    var new_energy = energy_i
        + params.z * resonance
        - params.w * energy_i;

    if (new_energy < 0.0) {
        new_energy = 0.0;
    }

    out_data[t_off + i] = new_theta;
    out_data[e_off + i] = new_energy;

    // omega не змінюємо
    out_data[o_off + i] = omega_i;
}
```

```rust
pub fn step_lattice(field: &mut Lattice, dt: f32, k: f32, alpha: f32, beta: f32) {
    let w = field.width;
    let h = field.height;

    let mut new_theta = field.theta.clone();
    let mut new_energy = field.energy.clone();

    for y in 0..h {
        for x in 0..w {
            let i = idx(x, y, w);

            let mut coupling = 0.0;
            let mut count = 0.0;

            for dy in -1..=1 {
                for dx in -1..=1 {
                    if dx == 0 && dy == 0 { continue; }

                    let nx = wrap(x as isize + dx, w);
                    let ny = wrap(y as isize + dy, h);

                    let j = idx(nx, ny, w);

                    coupling += (field.theta[j] - field.theta[i]).sin();
                    count += 1.0;
                }
            }

            coupling *= k / count;

            let dtheta = field.omega[i] + coupling;
            new_theta[i] = field.theta[i] + dtheta * dt;

            let resonance = coupling.abs();

            let mut e = field.energy[i]
                + alpha * resonance
                - beta * field.energy[i];

            if e < 0.0 { e = 0.0; }

            new_energy[i] = e;
        }
    }

    field.theta = new_theta;
    field.energy = new_energy;
}
```

    let x = id.x;
    let y = id.y;

    if (x >= meta.width || y >= meta.height) {
        return;
    }

    let i = idx(x, y);

    let theta_i = theta[i];
    let omega_i = omega[i];
    let energy_i = energy[i];

    var coupling: f32 = 0.0;
    var count: f32 = 0.0;

    for (var dy: i32 = -1; dy <= 1; dy = dy + 1) {
        for (var dx: i32 = -1; dx <= 1; dx = dx + 1) {

            if (dx == 0 && dy == 0) { continue; }

            let nx = wrap(i32(x) + dx, meta.width);
            let ny = wrap(i32(y) + dy, meta.height);

            let j = idx(nx, ny);

            coupling += sin(theta[j] - theta_i);
            count += 1.0;
        }

    coupling = (params.y / count) * coupling;

    let dtheta = omega_i + coupling;
    out_theta[i] = theta_i + dtheta * params.x;

    let resonance = abs(coupling);

    var e = energy_i
        + params.z * resonance
        - params.w * energy_i;

    if (e < 0.0) { e = 0.0; }
