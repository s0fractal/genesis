Чудовий план. Виділення парсера в окремий модуль `src/compiler/sigma_glyph.ts` —
це правильний архітектурний крок. Він інкапсулює всю логіку трансляції і
дозволить нам дуже легко тестувати граматику ізольовано від решти рушія.

Більше того, я одразу додав туди і **генератор (серіалізатор)**
`serializeSigmaGlyph`. Ти абсолютно правий: зворотна генерація критично
необхідна. Мутатор у `mutator.ts` зможе зчитувати чистий Glyph, мутувати його
(через AST), а потім серіалізувати назад у надщільний рядок для запису в новий
коміт чи файл `legacy_text_substrate`.

Ось повна реалізація модуля `src/compiler/sigma_glyph.ts`:

```typescript
import {
  IRBlock,
  IRCall,
  IRConst,
  IRIf,
  IRNode,
  IROp,
  IRVar,
} from "../quine.ts";

/**
 * OMEGA-64 | Σ-GLYPH Parser & Serializer
 * A dense, S-expression based pure lambda syntax for representing ASTs.
 */

// --- MAPS ---
const OP_TO_GLYPH: Record<string, string> = {
  add: "+",
  sub: "-",
  mul: "*",
  div: "/",
  and: "&",
  or: "|",
  xor: "^",
  shl: "<<",
  shr: ">>",
};

const GLYPH_TO_OP: Record<string, string> = Object.entries(OP_TO_GLYPH).reduce(
  (acc, [key, val]) => ({ ...acc, [val]: key }),
  {},
);

// --- TOKENIZER ---
function tokenize(glyph: string): string[] {
  return glyph
    .replace(/\(/g, " ( ")
    .replace(/\)/g, " ) ")
    .trim()
    .split(/\s+/)
    .filter((t) => t.length > 0);
}

// --- PARSER (Σ-GLYPH -> IRNode AST) ---
export function parseSigmaGlyph(glyphString: string): IRNode {
  const tokens = tokenize(glyphString);

  function parseNext(): IRNode {
    if (tokens.length === 0) {
      throw new Error("Σ-GLYPH Parsing Error: Unexpected end of expression.");
    }

    const token = tokens.shift()!;

    // 1. Scope / List
    if (token === "(") {
      const head = tokens.shift()!;
      const args: IRNode[] = [];

      while (tokens[0] !== ")") {
        if (tokens.length === 0) {
          throw new Error(
            "Σ-GLYPH Parsing Error: Missing closing parenthesis ')'.",
          );
        }
        args.push(parseNext());
      }
      tokens.shift(); // Consume ")"

      // Determine Node Kind from Head
      if (GLYPH_TO_OP[head]) {
        return { kind: "op", op: GLYPH_TO_OP[head] as IROp["op"], args };
      }

      if (head === "?") {
        if (args.length !== 3) {
          throw new Error(
            "Σ-GLYPH Parsing Error: '?' (If) requires exactly 3 arguments (cond, then, else).",
          );
        }
        return { kind: "if", cond: args[0], then: args[1], else: args[2] };
      }

      if (head === "do") {
        if (args.length < 1) {
          throw new Error(
            "Σ-GLYPH Parsing Error: 'do' (Block) requires at least a return expression.",
          );
        }
        const result = args.pop()!;
        return { kind: "block", statements: args, result };
      }

      if (head.startsWith("@")) {
        return { kind: "call", target: head.substring(1), args };
      }

      throw new Error(
        `Σ-GLYPH Parsing Error: Unknown operator or macro '${head}'.`,
      );
    }

    // 2. Atom: Constant
    // We use strict number parsing to handle negatives and floats (though WGSL prefers ints)
    if (!isNaN(Number(token))) {
      return { kind: "const", value: Number(token), bits: 32 };
    }

    // 3. Atom: Variable
    return { kind: "var", name: token, type: "i32" };
  }

  const ast = parseNext();

  if (tokens.length > 0) {
    throw new Error(
      `Σ-GLYPH Parsing Error: Stray tokens remaining after successful parse -> [${
        tokens.join(", ")
      }]`,
    );
  }

  return ast;
}

// --- SERIALIZER (IRNode AST -> Σ-GLYPH) ---
export function serializeSigmaGlyph(node: IRNode): string {
  switch (node.kind) {
    case "const":
      return node.value.toString();

    case "var":
      return node.name;

    case "op": {
      const glyphOp = OP_TO_GLYPH[node.op];
      const argsStr = node.args.map(serializeSigmaGlyph).join(" ");
      return `(${glyphOp} ${argsStr})`;
    }

    case "call": {
      const argsStr = node.args.map(serializeSigmaGlyph).join(" ");
      return `(@${node.target}${argsStr.length > 0 ? " " + argsStr : ""})`;
    }

    case "if": {
      const condStr = serializeSigmaGlyph(node.cond);
      const thenStr = serializeSigmaGlyph(node.then);
      const elseStr = serializeSigmaGlyph(node.else);
      return `(? ${condStr} ${thenStr} ${elseStr})`;
    }

    case "block": {
      const stmtsStr = node.statements.map(serializeSigmaGlyph).join(" ");
      const resStr = serializeSigmaGlyph(node.result);
      return `(do${stmtsStr.length > 0 ? " " + stmtsStr : ""} ${resStr})`;
    }

    default:
      // TypeScript exhaustiveness check fallback
      throw new Error(`Σ-GLYPH Serialization Error: Unknown IRNode kind.`);
  }
}
```
